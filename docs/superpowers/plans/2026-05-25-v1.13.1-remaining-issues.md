**Priority:** P1 (only remaining high-priority item)

**Problem:** Read locks are stored in an in-memory `Map<string, Set<string>>`. On crash or restart, all read locks vanish — creating a silent correctness gap where concurrent write operations could proceed without knowing about active readers.

**Files:**
- Modify: `packages/core/src/orchestration/conflict-resolver.ts`
- Modify: `packages/core/src/database/schema.ts`
- Modify: `packages/core/src/database/locks.ts`

**Current State:**
- Write locks already persist via `LocksRepository` in the `locks` table
- Read locks use `private readers = new Map<string, Set<string>>()`
- Constructor takes `RawDB` and uses `LocksRepository`
- Locks table: `path TEXT PRIMARY KEY, agent_id TEXT, expires_at INTEGER, created_at INTEGER`

```typescript
acquire(path: string, agentId: string, durationMs = 300_000, mode: 'read' | 'write' = 'write'): boolean {
  const now = Date.now();
  const existing = this.db.prepare(
    'SELECT * FROM locks WHERE path = ? AND expires_at > ? AND agent_id != ?'
  ).all(path, now, agentId) as any[];

  if (mode === 'read' && existing.some((l: any) => l.mode === 'write')) return false;

  this.db.prepare(
    'INSERT OR REPLACE INTO locks (path, agent_id, expires_at, created_at, mode) VALUES (?, ?, ?, ?, ?)'
  ).run(path, agentId, now + durationMs, now, mode);
  return true;
}
```

### Task 2: Paginate Audit Log verifyIntegrity()

**Priority:** P2

**Problem:** `verifyIntegrity()` does `SELECT * FROM audit_log ORDER BY sequence ASC` — loads entire table into memory. For production systems with thousands of entries, this is O(n) memory.

**Files:**
- Modify: `packages/core/src/resilience/audit-log.ts`

**Current Implementation:**
- Loads ALL entries, iterates sequentially verifying hash chain
- Hash input: `${id}:${agentId}:${action}:${detail}:${timestamp}:${sequence}:${prev_hash}`

**Implementation:**

- [ ] **Step 1: Extract hash computation into private method**

```typescript
private computeHash(entry: any, prevHash: string): string {
  const input = `${entry.id}:${entry.agent_id}:${entry.action}:${entry.detail}:${entry.timestamp}:${entry.sequence}:${prevHash}`;
  return createHash('sha256').update(input).digest('hex');
}
```

- [ ] **Step 2: Refactor verifyIntegrity to process in batches**

```typescript
verifyIntegrity(options?: { batchSize?: number; fromSequence?: number }): { valid: boolean; brokenAt?: string; lastVerifiedSequence: number } {
  const batchSize = options?.batchSize ?? 1000;
  let offset = options?.fromSequence ?? 0;
  let prevHash = '';

  if (offset > 0) {
    const prev = this.db.prepare('SELECT hash FROM audit_log WHERE sequence = ?').get(offset) as any;
    if (prev) prevHash = prev.hash;
  }

  while (true) {
    const batch = this.db.prepare(
      'SELECT * FROM audit_log WHERE sequence > ? ORDER BY sequence ASC LIMIT ?'
    ).all(offset, batchSize) as any[];

    if (batch.length === 0) break;

    for (const entry of batch) {
      const expectedHash = this.computeHash(entry, prevHash);
      if (entry.hash !== expectedHash || entry.prev_hash !== prevHash) {
        return { valid: false, brokenAt: entry.id, lastVerifiedSequence: offset };
      }
      prevHash = entry.hash;
      offset = entry.sequence;
    }
  }

  return { valid: true, lastVerifiedSequence: offset };
}
```

- [ ] **Step 4: Commit**

```bash

Processes 1000 entries at a time instead of loading entire table.
```

```typescript
// packages/core/src/events/event-store.ts
import type { RawDB } from '../database/types.js';
import type { WorkspaceEvent } from './types.js';

export class EventStore {
  constructor(private db: RawDB) {}

  append(event: WorkspaceEvent): number {
    const result = this.db.prepare(
      'INSERT INTO event_log (type, payload, timestamp) VALUES (?, ?, ?)'
    ).run(event.type, JSON.stringify(event), Date.now());
    return result.lastInsertRowid as number;
  }

  getUnreplayed(limit = 100): Array<{ id: number; event: WorkspaceEvent }> {
    const rows = this.db.prepare(
      'SELECT id, payload FROM event_log WHERE replayed = 0 ORDER BY id ASC LIMIT ?'
    ).all(limit) as any[];
    return rows.map(r => ({ id: r.id, event: JSON.parse(r.payload) }));
  }

  markReplayed(id: number): void {
    this.db.prepare('UPDATE event_log SET replayed = 1 WHERE id = ?').run(id);
  }

  getSince(afterId: number, type?: string, limit = 100): WorkspaceEvent[] {
    const sql = type
      ? 'SELECT payload FROM event_log WHERE id > ? AND type = ? ORDER BY id ASC LIMIT ?'
      : 'SELECT payload FROM event_log WHERE id > ? ORDER BY id ASC LIMIT ?';
    const params = type ? [afterId, type, limit] : [afterId, limit];
    const rows = this.db.prepare(sql).all(...params) as any[];
    return rows.map(r => JSON.parse(r.payload));
  }

  prune(olderThanMs: number): number {
    const cutoff = Date.now() - olderThanMs;
    const result = this.db.prepare(
      'DELETE FROM event_log WHERE timestamp < ? AND replayed = 1'
    ).run(cutoff);
    return result.changes;
  }
}
```

- [ ] **Step 3: Integrate EventStore into WorkspaceEventBus**

Add optional `store` to constructor. Persist before emitting:

```typescript
constructor(private store?: EventStore) {}

emit(event: WorkspaceEvent): void {
  if (this.store) this.store.append(event);
  // ...existing handler dispatch...
}

replay(): void {
  if (!this.store) return;
  const events = this.store.getUnreplayed();
  for (const { id, event } of events) {
    // Dispatch to handlers without re-persisting
    const handlers = this.handlers.get(event.type);
    if (handlers) for (const h of handlers) { try { h(event as any); } catch {} }
    this.store.markReplayed(id);
  }
}
```

```typescript
async generateBatch(texts: string[]): Promise<Float32Array[]> {
  if (!this.apiKey) throw new Error('Gemini API key required');
  const BATCH_SIZE = 100;
  const results: Float32Array[] = [];

  for (let i = 0; i < texts.length; i += BATCH_SIZE) {
    const batch = texts.slice(i, i + BATCH_SIZE);
    const controller = new AbortController();
    const timeout = setTimeout(() => controller.abort(), 30_000);
    try {
      const response = await fetch(
        'https://generativelanguage.googleapis.com/v1beta/models/text-embedding-004:batchEmbedContents',
        {
          method: 'POST',
          headers: { 'Content-Type': 'application/json', 'x-goog-api-key': this.apiKey },
          signal: controller.signal,
          body: JSON.stringify({
            requests: batch.map(text => ({ model: 'models/text-embedding-004', content: { parts: [{ text }] } })),
          }),
        }
      );
      const data = await response.json() as any;
      if (!data.embeddings) throw new Error(`Batch failed: ${JSON.stringify(data)}`);
      for (const emb of data.embeddings) results.push(new Float32Array(emb.values));
    } finally { clearTimeout(timeout); }
  }
  return results;
}
```

```typescript
// packages/core/src/metrics/exporter.ts
import type { MetricsSnapshot } from './collector.js';

export function toPrometheusText(snapshot: MetricsSnapshot, prefix = 'contextos'): string {
  const lines: string[] = [];
  for (const [name, value] of Object.entries(snapshot.counters)) {
    const m = `${prefix}_${name.replace(/\./g, '_')}_total`;
    lines.push(`# TYPE ${m} counter`, `${m} ${value}`);
  }
  for (const [name, hist] of Object.entries(snapshot.histograms)) {
    const m = `${prefix}_${name.replace(/\./g, '_')}`;
    lines.push(`# TYPE ${m} summary`);
    lines.push(`${m}{quantile="0.5"} ${hist.p50}`);
    lines.push(`${m}{quantile="0.95"} ${hist.p95}`);
    lines.push(`${m}{quantile="0.99"} ${hist.p99}`);
    lines.push(`${m}_sum ${hist.sum}`, `${m}_count ${hist.count}`);
  }
  for (const [name, value] of Object.entries(snapshot.gauges)) {
    const m = `${prefix}_${name.replace(/\./g, '_')}`;
    lines.push(`# TYPE ${m} gauge`, `${m} ${value}`);
  }
  return lines.join('\n') + '\n';
}
```
