```typescript
// workspace-mcp/src/tests/http-body-limit.test.ts
import assert from 'node:assert';
import { describe, it } from 'mocha';
import { Readable } from 'node:stream';

function collectBody(stream: Readable, maxBytes: number): Promise<Buffer> {
  return new Promise((resolve, reject) => {
    const chunks: Buffer[] = [];
    let totalBytes = 0;
    stream.on('data', (chunk: Buffer) => {
      totalBytes += chunk.length;
      if (totalBytes > maxBytes) {
        stream.destroy();
        reject(new Error('Request body too large'));
        return;
      }
      chunks.push(chunk);
    });
    stream.on('end', () => resolve(Buffer.concat(chunks)));
    stream.on('error', reject);
  });
}

describe('HTTP Body Stream Accumulation', () => {
  it('should reject body exceeding maxBytes', async () => {
    const oversizedData = Buffer.alloc(2_000_000, 'x');
    const stream = Readable.from([oversizedData]);
    await assert.rejects(
      collectBody(stream, 1_048_576),
      { message: 'Request body too large' }
    );
  });

  it('should accept body within maxBytes', async () => {
    const validData = Buffer.from('{"jsonrpc":"2.0","method":"ping"}');
    const stream = Readable.from([validData]);
    const result = await collectBody(stream, 1_048_576);
    assert.strictEqual(result.toString(), '{"jsonrpc":"2.0","method":"ping"}');
  });

  it('should handle chunked delivery', async () => {
    const chunk1 = Buffer.from('{"json');
    const chunk2 = Buffer.from('rpc":"2.0"}');
    const stream = Readable.from([chunk1, chunk2]);
    const result = await collectBody(stream, 1_048_576);
    assert.strictEqual(result.toString(), '{"jsonrpc":"2.0"}');
  });
});
```
