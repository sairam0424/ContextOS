// packages/core/src/tests/container-lifecycle.test.ts
import assert from 'node:assert';
import { describe, it } from 'mocha';

interface Disposable { dispose(): Promise<void> | void; }
interface Warmable { warmup(): Promise<void> | void; }

function isDisposable(obj: unknown): obj is Disposable {
  return typeof obj === 'object' && obj !== null && 'dispose' in obj && typeof (obj as any).dispose === 'function';
}

function isWarmable(obj: unknown): obj is Warmable {
  return typeof obj === 'object' && obj !== null && 'warmup' in obj && typeof (obj as any).warmup === 'function';
}

describe('Container Lifecycle', () => {
  it('should detect Disposable services', () => {
    const service = { dispose: async () => {} };
    assert.strictEqual(isDisposable(service), true);
  });

  it('should detect Warmable services', () => {
    const service = { warmup: async () => {} };
    assert.strictEqual(isWarmable(service), true);
  });

  it('should not flag plain objects as Disposable', () => {
    const service = { doWork: () => {} };
    assert.strictEqual(isDisposable(service), false);
  });

  it('start should call warmup on all warmable services', async () => {
    const warmupCalls: string[] = [];
    const services = [
      { name: 'a', warmup: async () => { warmupCalls.push('a'); } },
      { name: 'b' },
      { name: 'c', warmup: async () => { warmupCalls.push('c'); } },
    ];
    for (const s of services) {
      if (isWarmable(s)) await s.warmup();
    }
    assert.deepStrictEqual(warmupCalls, ['a', 'c']);
  });

  it('stop should call dispose on all disposable services in reverse order', async () => {
    const disposeCalls: string[] = [];
    const services = [
      { name: 'a', dispose: async () => { disposeCalls.push('a'); } },
      { name: 'b', dispose: async () => { disposeCalls.push('b'); } },
      { name: 'c', dispose: async () => { disposeCalls.push('c'); } },
    ];
    for (const s of [...services].reverse()) {
      if (isDisposable(s)) await s.dispose();
    }
    assert.deepStrictEqual(disposeCalls, ['c', 'b', 'a']);
  });
});
```

async start(): Promise<void> {
  for (const instance of this.instances.values()) {
    if (typeof instance === 'object' && instance !== null &&
        'warmup' in instance && typeof (instance as any).warmup === 'function') {
      await (instance as any).warmup();
    }
  }
}

async stop(): Promise<void> {
  const instances = [...this.instances.values()].reverse();
  for (const instance of instances) {
    if (typeof instance === 'object' && instance !== null &&
        'dispose' in instance && typeof (instance as any).dispose === 'function') {
      await (instance as any).dispose();
    }
  }
  this.instances.clear();
}

private persistState(agentId: string): void {
  if (!this.db) return;
  const state = this.getState(agentId);
  const record = this.errors.get(agentId);
  this.db.prepare(`INSERT OR REPLACE INTO circuit_breaker_state (id, state, tripped_at, error_count, last_error, updated_at)
    VALUES (?, ?, ?, ?, ?, ?)`).run(
    agentId, state, record?.trippedAt ?? null, record?.timestamps.length ?? 0, null, Date.now()
  );
}

private restoreState(agentId: string): void {
  if (!this.db || this.errors.has(agentId)) return;
  const row = this.db.prepare('SELECT * FROM circuit_breaker_state WHERE id = ?').get(agentId) as any;
  if (!row || row.state === 'closed') return;
  this.errors.set(agentId, {
    timestamps: Array(row.error_count).fill(row.tripped_at ?? Date.now()),
    trippedAt: row.tripped_at,
  });
}

// packages/core/src/tests/metrics-collector.test.ts
import assert from 'node:assert';
import { describe, it, beforeEach } from 'mocha';

class MetricsCollector {
  private counters = new Map<string, number>();
  private histograms = new Map<string, number[]>();
  private gauges = new Map<string, number>();
  increment(name: string, value = 1): void { this.counters.set(name, (this.counters.get(name) ?? 0) + value); }
  observe(name: string, value: number): void { const e = this.histograms.get(name) ?? []; e.push(value); this.histograms.set(name, e); }
  gauge(name: string, value: number): void { this.gauges.set(name, value); }
  snapshot() {
    const h: Record<string, any> = {};
    for (const [name, values] of this.histograms) {
      const sorted = [...values].sort((a, b) => a - b);
      h[name] = { count: sorted.length, sum: sorted.reduce((a, b) => a + b, 0), p50: sorted[Math.floor(sorted.length * 0.5)] ?? 0, p95: sorted[Math.floor(sorted.length * 0.95)] ?? 0, p99: sorted[Math.floor(sorted.length * 0.99)] ?? 0 };
    }
    return { counters: Object.fromEntries(this.counters), histograms: h, gauges: Object.fromEntries(this.gauges) };
  }
  reset(): void { this.counters.clear(); this.histograms.clear(); this.gauges.clear(); }
}

describe('MetricsCollector', () => {
  let collector: MetricsCollector;
  beforeEach(() => { collector = new MetricsCollector(); });

  it('should increment counters', () => {
    collector.increment('tasks.assigned');
    collector.increment('tasks.assigned');
    const snap = collector.snapshot();
    assert.strictEqual(snap.counters['tasks.assigned'], 2);
  });

  it('should track histogram observations', () => {
    collector.observe('search.latency_ms', 10);
    collector.observe('search.latency_ms', 20);
    collector.observe('search.latency_ms', 30);
    const snap = collector.snapshot();
    assert.strictEqual(snap.histograms['search.latency_ms'].count, 3);
    assert.strictEqual(snap.histograms['search.latency_ms'].sum, 60);
  });

  it('should set gauges', () => {
    collector.gauge('agents.active', 5);
    collector.gauge('agents.active', 3);
    assert.strictEqual(collector.snapshot().gauges['agents.active'], 3);
  });

  it('should reset all metrics', () => {
    collector.increment('x'); collector.observe('y', 1); collector.gauge('z', 1);
    collector.reset();
    const snap = collector.snapshot();
    assert.deepStrictEqual(snap.counters, {});
    assert.deepStrictEqual(snap.histograms, {});
    assert.deepStrictEqual(snap.gauges, {});
  });
});
```

describe('Capability-Based Routing', () => {
  it('should filter agents by required capabilities', () => {
    const agents = [
      { id: 'a1', capabilities: ['search', 'index'], load: 2 },
      { id: 'a2', capabilities: ['search'], load: 1 },
      { id: 'a3', capabilities: ['repair', 'index'], load: 0 },
    ];
    const required = ['search', 'index'];
    const candidates = agents.filter(a => required.every(cap => a.capabilities.includes(cap)));

    const required: string[] = [];
    const candidates = required.length === 0 ? agents : agents.filter(a => required.every(cap => a.capabilities.includes(cap)));
    assert.strictEqual(candidates.length, 2);
  });

  it('should pick least-loaded from matching candidates', () => {
    const candidates = [{ id: 'a1', load: 5 }, { id: 'a2', load: 1 }, { id: 'a3', load: 3 }];

// packages/core/src/tests/retry-backoff.test.ts
import assert from 'node:assert';
import { describe, it } from 'mocha';

interface RetryConfig { maxRetries: number; baseDelayMs: number; maxDelayMs: number; backoffMultiplier: number; }

function calculateDelay(retryCount: number, config: RetryConfig): number {
  return Math.min(config.baseDelayMs * (config.backoffMultiplier ** retryCount), config.maxDelayMs);
}

describe('Retry with Exponential Backoff', () => {
  const config: RetryConfig = { maxRetries: 3, baseDelayMs: 1000, maxDelayMs: 30000, backoffMultiplier: 2 };

  it('should calculate exponential delays', () => {
    assert.strictEqual(calculateDelay(1, config), 2000);
    assert.strictEqual(calculateDelay(2, config), 4000);
    assert.strictEqual(calculateDelay(3, config), 8000);
  });

  it('should cap at maxDelayMs', () => {
    assert.strictEqual(calculateDelay(10, config), 30000);
  });

  it('should use base delay for first retry', () => {
    assert.strictEqual(calculateDelay(0, config), 1000);
  });
});
