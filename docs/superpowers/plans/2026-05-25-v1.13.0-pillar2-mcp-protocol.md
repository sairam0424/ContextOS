// workspace-mcp/src/prompts/daily-standup.ts
import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
import { z } from 'zod';
import { samplingService } from '@context-os/core';

export function registerDailyStandup(server: McpServer): void {
  server.prompt(
    'daily-standup',
    'Generates a daily context summary for the current workspace',
    { date: z.string().optional().describe('ISO date to summarize (defaults to today)') },
    async ({ date }) => {
      const targetDate = date ?? new Date().toISOString().slice(0, 10);
      const pulse = await samplingService.getPulse();
      const summary = [
        `# Daily Standup — ${targetDate}`,
        '',
        `## Workspace Health`,
        `- Health Score: ${pulse.healthScore ?? 'N/A'}`,
        `- Total Documents: ${pulse.totalDocuments ?? 0}`,
        `- Recent Changes: ${pulse.recentChanges ?? 0}`,
        '',
        `## Summary`,
        `Provide a brief update on what was accomplished, what's in progress, and any blockers.`,
      ].join('\n');

      return {
        messages: [{ role: 'user', content: { type: 'text', text: summary } }],
      };
    }
  );
}
```

import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
import { z } from 'zod';
import { missionService } from '@context-os/core';

export function registerMissionBrief(server: McpServer): void {
  server.prompt(
    'mission-brief',
    'Summarizes active mission state and next actions',
    {
      missionId: z.string().optional().describe('Specific mission ID (defaults to active mission)'),
    },
    async ({ missionId }) => {

      const target = missionId
        ? missions.find((m: any) => m.id === missionId)

          messages: [{ role: 'user', content: { type: 'text', text: 'No active missions found.' } }],
        };

      const text = [
        `# Mission Brief: ${target.title}`,
        '',
        `**Status:** ${target.status}`,
        `**Created:** ${new Date(target.createdAt).toISOString()}`,
        '',
        `## Objective`,
        target.description ?? 'No description provided.',
      ].join('\n');

      return {
        messages: [{ role: 'user', content: { type: 'text', text } }],
      };
    }
  );
}

// workspace-mcp/src/logging.ts
import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';

export type LogLevel = 'debug' | 'info' | 'warning' | 'error';

let serverInstance: McpServer | null = null;

export function setServerInstance(server: McpServer): void {
  serverInstance = server;
}

export function logToClient(level: LogLevel, data: string | Record<string, unknown>): void {
  if (!serverInstance) return;
  try {
    serverInstance.server.sendLoggingMessage({ level, data });
  } catch {
    // Client may not support logging — silently ignore
  }
}

export function wrapToolHandler<T extends (...args: any[]) => any>(
  toolName: string,
  handler: T
): T {
  return (async (...args: any[]) => {
    logToClient('info', { tool: toolName, action: 'invoked', timestamp: Date.now() });
    try {
      const result = await handler(...args);
      logToClient('debug', { tool: toolName, action: 'completed', timestamp: Date.now() });
      return result;
    } catch (error: any) {
      logToClient('error', { tool: toolName, action: 'failed', error: error.message, timestamp: Date.now() });
      throw error;
    }
  }) as unknown as T;
}
```

```typescript
// workspace-mcp/src/subscriptions.ts
import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';

export class SubscriptionManager {
  private subscriptions = new Map<string, Set<string>>();
  private server: McpServer | null = null;

  attach(server: McpServer): void {
    this.server = server;
  }

  subscribe(uri: string, sessionId: string): void {
    if (!this.subscriptions.has(uri)) {
      this.subscriptions.set(uri, new Set());
    }
    this.subscriptions.get(uri)!.add(sessionId);
  }

  unsubscribe(uri: string, sessionId: string): void {
    const sessions = this.subscriptions.get(uri);
    if (sessions) {
      sessions.delete(sessionId);
      if (sessions.size === 0) {
        this.subscriptions.delete(uri);
      }
    }
  }

  notifyResourceUpdated(uri: string): void {
    if (!this.server) return;
    const sessions = this.subscriptions.get(uri);
    if (!sessions || sessions.size === 0) return;
    try {
      this.server.server.sendResourceUpdated({ uri });
    } catch {
      // Client may not support subscriptions
    }
  }

  getSubscriberCount(uri: string): number {
    return this.subscriptions.get(uri)?.size ?? 0;
  }

  cleanup(sessionId: string): void {
    for (const [uri, sessions] of this.subscriptions) {
      sessions.delete(sessionId);
      if (sessions.size === 0) {
        this.subscriptions.delete(uri);
      }
    }
  }
}

export const subscriptionManager = new SubscriptionManager();
```

// workspace-mcp/src/tools/graph-query.ts
import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
import { z } from 'zod';
import { knowledgeGraphService } from '@context-os/core';
import { handleToolError } from '../utils.js';

export function registerGraphQueryTool(server: McpServer): void {
  server.tool(
    'graph_query',
    {
      startNode: z.string().describe('Node ID or path to start traversal from'),
      direction: z.enum(['outbound', 'inbound', 'both']).default('both').describe('Traversal direction'),
      maxDepth: z.number().min(1).max(5).default(2).describe('Maximum traversal depth'),
      minWeight: z.number().min(0).max(1).default(0.1).describe('Minimum edge weight threshold'),
      limit: z.number().min(1).max(100).default(20).describe('Maximum results to return'),
      nodeType: z.string().optional().describe('Filter results by node type'),
    },
    async ({ startNode, direction, maxDepth, minWeight, limit, nodeType }) => {
      try {
        const graph = knowledgeGraphService.getGraph();
        const results = graph.getAffinities(startNode, minWeight, limit);

        const filtered = nodeType
          ? results.filter((r: any) => r.type === nodeType)
          : results;

        const formatted = filtered.map((r: any) => ({
          node: r.target,
          weight: r.weight,
          type: r.type ?? 'unknown',
        }));

        return {
          content: [{
            type: 'text' as const,
            text: JSON.stringify({ startNode, direction, maxDepth, results: formatted }, null, 2),
          }],
        };
      } catch (error: any) {
        return handleToolError(error, 'graph_query');
      }
    }
  );
}
```

// workspace-mcp/src/tools/workspace-notify.ts
import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
import { z } from 'zod';
import { MessageBus, createDefaultContainer, TOKENS } from '@context-os/core';
import { handleToolError } from '../utils.js';

export function registerWorkspaceNotifyTool(server: McpServer): void {
  server.tool(
    'workspace_notify',
    {
      action: z.enum(['send', 'read', 'broadcast']).describe('Message action'),
      topic: z.string().optional().describe('Message topic/channel'),
      targetAgentId: z.string().optional().describe('Target agent ID (for send action)'),
      content: z.string().optional().describe('Message content (for send/broadcast)'),
      since: z.string().optional().describe('ISO timestamp to read messages since'),
      agentId: z.string().optional().describe('Agent ID reading messages (for read action)'),
    },
    async ({ action, topic, targetAgentId, content, since, agentId }) => {
      try {
        const container = createDefaultContainer();
        const messageBus = container.resolve<MessageBus>(TOKENS.MessageBus);

        switch (action) {
          case 'send': {
            if (!targetAgentId || !content) {
              return {
                content: [{ type: 'text' as const, text: 'Error: send requires targetAgentId and content' }],
                isError: true,
              };
            }
            const messageId = messageBus.send({
              from: agentId ?? 'mcp-client',
              to: targetAgentId,
              intent: topic ?? 'notification',
              payload: { content },
            });
            return {
              content: [{ type: 'text' as const, text: `Message sent (id: ${messageId})` }],
            };
          }

          case 'broadcast': {
            if (!content) {
              return {
                content: [{ type: 'text' as const, text: 'Error: broadcast requires content' }],
                isError: true,
              };
            }
            const messageId = messageBus.send({
              from: agentId ?? 'mcp-client',
              to: '*',
              intent: topic ?? 'broadcast',
              payload: { content },
            });
            return {
              content: [{ type: 'text' as const, text: `Broadcast sent (id: ${messageId})` }],
            };
          }

          case 'read': {
            if (!agentId) {
              return {
                content: [{ type: 'text' as const, text: 'Error: read requires agentId' }],
                isError: true,
              };
            }
            const messages = messageBus.getUndelivered(agentId);
            const filtered = since
              ? messages.filter((m: any) => m.timestamp >= new Date(since).getTime())
              : messages;
            return {
              content: [{
                type: 'text' as const,
                text: JSON.stringify(filtered, null, 2),
              }],
            };
          }
        }
      } catch (error: any) {
        return handleToolError(error, 'workspace_notify');
      }
    }
  );
}
```

    assert.strictEqual(isValid, true);
  });

  it('should validate read action requires agentId', () => {
    const input = { action: 'read' as const, agentId: 'agent-1' };
    const isValid = input.action === 'read' && !!input.agentId;
    assert.strictEqual(isValid, true);
  });

  it('should support all three actions', () => {
    const actions = ['send', 'read', 'broadcast'];
    assert.strictEqual(actions.length, 3);
    actions.forEach(a => assert.ok(['send', 'read', 'broadcast'].includes(a)));
